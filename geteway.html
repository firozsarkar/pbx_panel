#!/bin/bash # ═══════════════════════════════════════════════════════ # FreeSWITCH Install Script - Debian 11/12 / Ubuntu 22 # ═══════════════════════════════════════════════════════ STEP 1: Add FreeSWITCH Repository apt-get update && apt-get install -y gnupg2 wget lsb-release wget -O /usr/share/keyrings/freeswitch-archive-keyring.gpg \ https://files.freeswitch.org/repo/deb/debian-release/fsstretch-archive-keyring.gpg echo "deb [signed-by=/usr/share/keyrings/freeswitch-archive-keyring.gpg] \ https://files.freeswitch.org/repo/deb/debian-release/ \ $(lsb_release -sc) main" > /etc/apt/sources.list.d/freeswitch.list STEP 2: Install FreeSWITCH apt-get update apt-get install -y freeswitch-meta-default STEP 3: Enable & Start systemctl enable freeswitch systemctl start freeswitch systemctl status freeswitch STEP 4: Test with fs_cli fs_cli # Then type: sofia status STEP 5: Install Nginx for Web UI apt-get install -y nginx php-fpm php-cli systemctl enable nginx && systemctl start nginx STEP 6: Copy this gateway.html to web root cp gateway.html /var/www/html/gateway.html chmod 644 /var/www/html/gateway.html # Access: http://YOUR-VPS-IP/gateway.html
# ═══════════════════════════════════════ # Gateway Configuration Guide # ═══════════════════════════════════════ STEP 1: Go to the directory cd /etc/freeswitch/sip_profiles/external/ STEP 2: Create gateway XML (example: mytrunk.xml) nano mytrunk.xml # Paste the XML generated from "XML Generator" tab STEP 3: Reload FreeSWITCH config fs_cli -x "reloadxml" fs_cli -x "sofia profile external rescan" STEP 4: Check gateway registration fs_cli -x "sofia status gateway mytrunk" STEP 5: Test outbound call fs_cli -x "originate sofia/gateway/mytrunk/+8801XXXXXXXXX &echo" STEP 6: Dialplan for outbound (default.xml) nano /etc/freeswitch/dialplan/default.xml # Add extension block from Dialplan Generator STEP 7: Check logs if issue tail -f /var/log/freeswitch/freeswitch.log | grep -i gateway tail -f /var/log/freeswitch/freeswitch.log | grep -i sofia
# ═══════════════════════════════════════ # Deploy this Web UI on VPS # ═══════════════════════════════════════ METHOD 1: Direct HTML (Simplest) cp /path/to/gateway.html /var/www/html/gateway.html # Access: http://YOUR_IP/gateway.html METHOD 2: With Nginx Virtual Host nano /etc/nginx/sites-available/freeswitch-ui server { listen 80; server_name your-domain.com; root /var/www/html; index gateway.html; location / { try_files $uri $uri/ =404; } } ln -s /etc/nginx/sites-available/freeswitch-ui \ /etc/nginx/sites-enabled/ nginx -t && systemctl reload nginx METHOD 3: With SSL (certbot) apt install certbot python3-certbot-nginx -y certbot --nginx -d your-domain.com Protect with basic auth (optional) apt install apache2-utils -y htpasswd -c /etc/nginx/.htpasswd admin # Add to nginx config: # auth_basic "FreeSWITCH UI"; # auth_basic_user_file /etc/nginx/.htpasswd;
# ═══════════════════════════════════════ # Important Tips & Troubleshooting # ═══════════════════════════════════════ Firewall Rules (UFW) ufw allow 5060/udp # SIP ufw allow 5060/tcp # SIP TCP ufw allow 5061/tcp # SIP TLS ufw allow 16384:32768/udp # RTP ufw allow 8021/tcp # ESL (restrict to trusted IP!) ufw allow 80/tcp # Web UI HTTP ufw allow 443/tcp # Web UI HTTPS Gateway Not Registering? 1. Check: fs_cli -x "sofia loglevel all 9" One-Way Audio Fix 1. Set rtp-ip to server public IP 2. Set ext-rtp-ip to public IP 3. Open RTP ports: 16384-32768 UDP 4.
